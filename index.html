// ========================================== // Canvas塗りつぶし＆赤枠ハイライト用 座標データ構造 // ========================================== var markW = 12; // ご指定の横幅 var markH = 25; // ご指定の縦幅 // 基本用（緑）の座標を自動計算で構築 var greenCoords = { venue: {}, race: {}, shikibetsu: {}, rows: [], highlight: { venue: {x: 20, y: 30, w: 140, h: 50}, race: {x: 170, y: 30, w: 140, h: 50} } }; }); // ②レース番号 (2行6列) for (var i = 1; i <= 12; i++) { var col = (i - 1) % 6; var row = Math.floor((i - 1) / 6); greenCoords.race[i.toString()] = { x: 410 + (col * 22), y: 55 + (row * 35), w: markW, h: markH }; } // ③式別 (1行8列) var types = ['tansho', 'fukusho', 'wakuren', 'umaren', 'umatan', 'wide', 'sanrenfuku', 'sanrentan']; types.forEach(function(t, i) { greenCoords.shikibetsu[t] = { x: 575 + (i * 24), y: 72, w: markW, h: markH }; }); // ④馬番・枠番、⑤金額、⑥単位 (4行分ループ生成) for (var r = 0; r < 4; r++) { var rowObj = { num1: {}, num2: {}, num3: {}, amountNum: {}, amountUnit: {} }; var baseY = 135 + (r * 75); // 各行の基準Y座標 // 馬番（1〜18）を生成する関数 (2行9列) function generateNums(startX) { var nums = {}; for (var n = 1; n <= 18; n++) { var c = (n - 1) % 9; var ro = Math.floor((n - 1) / 9); nums[n.toString()] = { x: startX + (c * 17), y: baseY + (ro * 33), w: markW, h: markH }; } return nums; } rowObj.num1 = generateNums(55); // 1着・1頭目 rowObj.num2 = generateNums(225); // 2着・2頭目 rowObj.num3 = generateNums(395); // 3着・3頭目 // 金額 (2行4列: 上段は10,20,30,5 / 下段は4,3,2,1) var amtValues = [10, 20, 30, 5, 4, 3, 2, 1]; amtValues.forEach(function(val, idx) { var c = idx % 4; var ro = Math.floor(idx / 4); rowObj.amountNum[val.toString()] = { x: 575 + (c * 20), y: baseY + (ro * 33), w: markW, h: markH }; }); // 単位 (1行3列: 万, 千, 百) var units = [10000, 1000, 100]; units.forEach(function(val, idx) { rowObj.amountUnit[val.toString()] = { x: 675 + (idx * 20), y: baseY, w: markW, h: markH }; }); greenCoords.rows.push(rowObj); } // 最終的なデータ構造の定義 var markSheetCoords = { green: greenCoords, // 自動計算した緑の座標セットを代入 redBox: { shikibetsu: { 'umaren': {x: 100, y: 50, w: markW, h: markH} }, highlight: { venue: {x: 20, y: 30, w: 140, h: 50}, race: {x: 170, y: 30, w: 140, h: 50} } }, redForm: { shikibetsu: {}, highlight: { venue: {x: 20, y: 30, w: 140, h: 50}, race: {x: 170, y: 30, w: 140, h: 50} } }, blueFuku: { highlight: { venue: {x: 20, y: 30, w: 140, h: 50}, race: {x: 170, y: 30, w: 140, h: 50} } }, blueTan: { highlight: { venue: {x: 20, y: 30, w: 140, h: 50}, race: {x: 170, y: 30, w: 140, h: 50} } } };
